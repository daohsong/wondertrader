
#1. 确定CMake的最低版本需求
cmake_minimum_required(VERSION 3.20)

#2. 确定工程名
project(CTPOptLoader LANGUAGES CXX)


#7. 添加源码
SET(SRC  
	${PROJECT_SOURCE_DIR}/CTPOptLoader.cpp
	${PROJECT_SOURCE_DIR}/TraderSpi.cpp
	${PROJECT_SOURCE_DIR}/TraderSpi.h
)


SET(LIBS
	${FS_LIB}
    WTSUtils
	WT::Platform
	${WT_ICONV_LIB}
)

IF(WIN32)
LIST(APPEND LIBS iconv)
ENDIF()

#9. 生成可执行文件
ADD_LIBRARY(CTPOptLoader SHARED ${SRC})
TARGET_LINK_LIBRARIES(CTPOptLoader PRIVATE WT::fmt WT::RapidJSON PUBLIC WT::Vendor::CTPOpt)
IF (MSVC)
ELSE()
	TARGET_LINK_LIBRARIES(CTPOptLoader PUBLIC ${LIBS})
ENDIF ()

IF (MSVC)
ELSE()
	SET_TARGET_PROPERTIES(CTPOptLoader PROPERTIES
		CXX_VISIBILITY_PRESET hidden
		C_VISIBILITY_PRESET hidden
		VISIBILITY_INLINES_HIDDEN 1
        BUILD_RPATH_USE_ORIGIN ON)
ENDIF ()
