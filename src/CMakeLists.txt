FUNCTION(WT_CTP_CHECK_LIBRARY_ARCH WT_LIBRARY WT_OUTPUT)
	SET(WT_CTP_ARCH_OK TRUE)
	IF(WT_MACOS_ARM64 AND WT_LIBRARY)
		IF(NOT EXISTS "${WT_LIBRARY}")
			SET(WT_CTP_ARCH_OK FALSE)
		ELSE()
			EXECUTE_PROCESS(
				COMMAND lipo -archs "${WT_LIBRARY}"
				OUTPUT_VARIABLE WT_CTP_ARCHS
				ERROR_VARIABLE WT_CTP_ARCH_ERROR
				RESULT_VARIABLE WT_CTP_ARCH_RESULT
				OUTPUT_STRIP_TRAILING_WHITESPACE
				ERROR_STRIP_TRAILING_WHITESPACE
			)
			IF(NOT WT_CTP_ARCH_RESULT EQUAL 0)
				SET(WT_CTP_ARCH_OK FALSE)
			ELSE()
				STRING(REGEX MATCH "(^|[ \t])arm64([ \t]|$)" WT_CTP_HAS_ARM64 "${WT_CTP_ARCHS}")
				IF(NOT WT_CTP_HAS_ARM64)
					SET(WT_CTP_ARCH_OK FALSE)
				ENDIF()
			ENDIF()
		ENDIF()
	ENDIF()
	SET(${WT_OUTPUT} ${WT_CTP_ARCH_OK} PARENT_SCOPE)
ENDFUNCTION()
