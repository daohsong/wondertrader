#1. 确定CMake的最低版本需求
CMAKE_MINIMUM_REQUIRED(VERSION 3.20)

#2. 确定工程名
PROJECT(TraderCTPMini LANGUAGES CXX)

SET(SRC  
	${PROJECT_SOURCE_DIR}/TraderCTPMini.cpp
	${PROJECT_SOURCE_DIR}/TraderCTPMini.h
)


ADD_LIBRARY(TraderCTPMini SHARED ${SRC})
TARGET_LINK_LIBRARIES(TraderCTPMini PRIVATE WT::fmt PUBLIC WT::Vendor::CTPMini)

IF(MSVC)
ELSE()
	SET(LIBS
		WT::Platform
		${FS_LIB}
		)
	TARGET_LINK_LIBRARIES(TraderCTPMini PUBLIC ${LIBS})
ENDIF()

IF (MSVC)
ELSE()
	SET_TARGET_PROPERTIES(TraderCTPMini PROPERTIES
		CXX_VISIBILITY_PRESET hidden
		C_VISIBILITY_PRESET hidden
		VISIBILITY_INLINES_HIDDEN 1
        BUILD_RPATH_USE_ORIGIN ON)
ENDIF ()
