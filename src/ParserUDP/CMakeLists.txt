
#1. 确定CMake的最低版本需求
CMAKE_MINIMUM_REQUIRED(VERSION 3.20)

#2. 确定工程名
PROJECT(ParserUDP LANGUAGES CXX)

SET(SRC  
	${PROJECT_SOURCE_DIR}/ParserUDP.cpp
	${PROJECT_SOURCE_DIR}/ParserUDP.h
)


ADD_LIBRARY(ParserUDP SHARED ${SRC})
TARGET_LINK_LIBRARIES(ParserUDP PRIVATE WT::fmt)

IF(MSVC)
	SET(LIBS
		ws2_32
	)
ELSE()
	IF (WIN32)
		SET(LIBS
			WT::BoostThread
			ws2_32
		)
	ELSE(UNIX)
		SET(LIBS
			WT::BoostThread
		)
	ENDIF()
ENDIF()
TARGET_LINK_LIBRARIES(ParserUDP PUBLIC ${LIBS})

IF (MSVC)
ELSE()
	SET_TARGET_PROPERTIES(ParserUDP PROPERTIES
		CXX_VISIBILITY_PRESET hidden
		C_VISIBILITY_PRESET hidden
		VISIBILITY_INLINES_HIDDEN 1
        BUILD_RPATH_USE_ORIGIN ON)
ENDIF ()
