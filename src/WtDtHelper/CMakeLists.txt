#1. 确定CMake的最低版本需求
CMAKE_MINIMUM_REQUIRED(VERSION 3.20)

#2. 确定工程名
PROJECT(WtDtHelper LANGUAGES CXX)

SET(LIBRARY_OUTPUT_PATH ${CMAKE_BINARY_DIR}/build_${PLATFORM}/${CMAKE_BUILD_TYPE}/bin/WtDtPorter)

#7. 添加源码
file(GLOB SRCS *.cpp)
file(GLOB HDRS *.h)

ADD_LIBRARY(WtDtHelper SHARED ${SRCS} ${HDRS})
TARGET_LINK_LIBRARIES(WtDtHelper PRIVATE WT::fmt WT::RapidJSON)

SET(LIBS
	WTSUtils
    WTSTools
)
IF (MSVC)
ELSE (GNUCC)
	LIST(APPEND LIBS
		WT::Platform
		${FS_LIB}
		WT::BoostFilesystem
		)
ENDIF()

TARGET_LINK_LIBRARIES(WtDtHelper PUBLIC ${LIBS})
IF (MSVC)
ELSE (GNUCC)
	SET_TARGET_PROPERTIES(WtDtHelper PROPERTIES
		CXX_VISIBILITY_PRESET hidden
		VISIBILITY_INLINES_HIDDEN 1
        LINK_FLAGS_RELEASE -s)
ENDIF ()
