#1. 确定CMake的最低版本需求
CMAKE_MINIMUM_REQUIRED(VERSION 3.20)

#2. 确定工程名
PROJECT(WtUftRunner LANGUAGES CXX)

SET(EXECUTABLE_OUTPUT_PATH ${CMAKE_BINARY_DIR}/build_${PLATFORM}/${CMAKE_BUILD_TYPE}/bin/WtUftRunner)

#7. 添加源码
file(GLOB SRCS *.cpp)
file(GLOB HDRS *.h)

IF(MSVC)
	list (APPEND SRCS ../Common/mdump.cpp)
ENDIF()


SET(LIBS
	WtUftCore
	WTSTools
    WTSUtils
	)

IF (MSVC)
	LIST(APPEND LIBS ws2_32)
ELSE(GNUCC)
	LIST(APPEND LIBS
		WT::Platform
		WT::BoostFilesystem
		WT::BoostThread
		${FS_LIB})
	IF(WIN32)
		LIST(APPEND LIBS
			ws2_32 iconv)
	ENDIF()
ENDIF()

ADD_EXECUTABLE(WtUftRunner ${SRCS} ${HDRS})
TARGET_LINK_LIBRARIES(WtUftRunner PRIVATE WT::fmt)
TARGET_LINK_LIBRARIES(WtUftRunner PRIVATE ${LIBS})

IF (MSVC)
ELSE (GNUCC)
	SET_TARGET_PROPERTIES(WtUftRunner PROPERTIES
        LINK_FLAGS_RELEASE -s)
ENDIF ()

file(MAKE_DIRECTORY ${EXECUTABLE_OUTPUT_PATH}/parsers/)
file(MAKE_DIRECTORY ${EXECUTABLE_OUTPUT_PATH}/traders/)

SET(WT_UFT_RUNNER_POST_BUILD_DEPENDS )
SET(WT_UFT_RUNNER_POST_BUILD_COMMANDS )

IF(TARGET TraderCTP)
	LIST(APPEND WT_UFT_RUNNER_POST_BUILD_DEPENDS TraderCTP)
	LIST(APPEND WT_UFT_RUNNER_POST_BUILD_COMMANDS
	COMMAND ${CMAKE_COMMAND} -E
	copy ${CMAKE_BINARY_DIR}/build_${PLATFORM}/${CMAKE_BUILD_TYPE}/bin/${PREFIX}TraderCTP${SUFFIX} ${EXECUTABLE_OUTPUT_PATH}/traders/
	)
ENDIF()

IF(TARGET ParserCTP)
	LIST(APPEND WT_UFT_RUNNER_POST_BUILD_DEPENDS ParserCTP)
	LIST(APPEND WT_UFT_RUNNER_POST_BUILD_COMMANDS
	COMMAND ${CMAKE_COMMAND} -E
	copy ${CMAKE_BINARY_DIR}/build_${PLATFORM}/${CMAKE_BUILD_TYPE}/bin/${PREFIX}ParserCTP${SUFFIX} ${EXECUTABLE_OUTPUT_PATH}/parsers/
	)
ENDIF()

IF(WT_UFT_RUNNER_POST_BUILD_COMMANDS)
	add_dependencies(WtUftRunner ${WT_UFT_RUNNER_POST_BUILD_DEPENDS})
	add_custom_command(TARGET WtUftRunner POST_BUILD
	${WT_UFT_RUNNER_POST_BUILD_COMMANDS}
	COMMENT "depends modules copied")
ENDIF()
