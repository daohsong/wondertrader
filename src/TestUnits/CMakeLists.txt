
#1. 确定CMake的最低版本需求
CMAKE_MINIMUM_REQUIRED(VERSION 3.20)

#2. 确定工程名
PROJECT(TestUnits LANGUAGES CXX)

SET(EXECUTABLE_OUTPUT_PATH ${CMAKE_BINARY_DIR}/build_${PLATFORM}/${CMAKE_BUILD_TYPE}/bin/TestUnits)

#7. 添加源码
file(GLOB SRCS *.cpp ./gtest/*.cc)

SET(LIBS
	WtDtCore
    WTSTools
	WTSUtils
    WtShareHelper)
IF (MSVC)
ELSE(GNUCC)
    LIST(APPEND LIBS ${WT_THREAD_LIB} ${FS_LIB})
	IF(WIN32)
		LIST(APPEND LIBS iconv)
	ENDIF()
ENDIF()

LIST(APPEND INCS "./gtest")

INCLUDE_DIRECTORIES(${INCS})
LINK_DIRECTORIES(${LNKS})

ADD_EXECUTABLE(TestUnits ${SRCS})
TARGET_LINK_LIBRARIES(TestUnits ${LIBS})

ADD_TEST(
	NAME WonderTrader.TestUnits
	COMMAND ${CMAKE_COMMAND}
		-DWT_TEST_EXECUTABLE=$<TARGET_FILE:TestUnits>
		-DWT_TEST_SANDBOX_ROOT=${CMAKE_CURRENT_BINARY_DIR}/sandboxes
		-P ${CMAKE_CURRENT_LIST_DIR}/RunTestInSandbox.cmake
)
SET_TESTS_PROPERTIES(WonderTrader.TestUnits PROPERTIES
	LABELS unit
	TIMEOUT 180
)

IF (MSVC)
ELSE (GNUCC)
	SET_TARGET_PROPERTIES(TestUnits PROPERTIES
        LINK_FLAGS_RELEASE -s)
ENDIF ()
