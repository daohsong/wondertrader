
#1. 确定CMake的最低版本需求
CMAKE_MINIMUM_REQUIRED(VERSION 3.20)

#2. 确定工程名
PROJECT(WtCtaStraFact LANGUAGES CXX)

#7. 添加源码
file(GLOB SRCS "*.cpp")
file(GLOB HDRS "*.h")

SET(LIBRARY_OUTPUT_PATH ${CMAKE_BINARY_DIR}/build_${PLATFORM}/${CMAKE_BUILD_TYPE}/bin)

ADD_LIBRARY(WtCtaStraFact SHARED ${SRCS} ${HDRS})
TARGET_LINK_LIBRARIES(WtCtaStraFact PRIVATE WT::fmt WT::Boost)

SET(LIBS
	${FS_LIB}
	WT::Platform
)

IF (WIN32)
ELSE (UNIX)
	TARGET_LINK_LIBRARIES(WtCtaStraFact PUBLIC ${LIBS})
	SET_TARGET_PROPERTIES(WtCtaStraFact PROPERTIES
		CXX_VISIBILITY_PRESET hidden
		C_VISIBILITY_PRESET hidden
		VISIBILITY_INLINES_HIDDEN 1
        LINK_FLAGS_RELEASE -s)
ENDIF ()
