
#1. 确定CMake的最低版本需求
CMAKE_MINIMUM_REQUIRED(VERSION 3.20)

#2. 确定工程名
PROJECT(ParserFemas LANGUAGES CXX)

SET(SRC  
	${PROJECT_SOURCE_DIR}/ParserFemas.cpp
	${PROJECT_SOURCE_DIR}/ParserFemas.h
)


ADD_LIBRARY(ParserFemas SHARED ${SRC})
TARGET_LINK_LIBRARIES(ParserFemas PRIVATE WT::fmt PUBLIC WT::Vendor::Femas)

IF(MSVC)
ELSE()
	SET(LIBS
		${FS_LIB}
		WT::Platform
	)
	TARGET_LINK_LIBRARIES(ParserFemas PUBLIC ${LIBS})
ENDIF()

IF (MSVC)
ELSE()
	SET_TARGET_PROPERTIES(ParserFemas PROPERTIES
		CXX_VISIBILITY_PRESET hidden
		C_VISIBILITY_PRESET hidden
		VISIBILITY_INLINES_HIDDEN 1
        BUILD_RPATH_USE_ORIGIN ON)
ENDIF ()
