
#1. 确定CMake的最低版本需求
CMAKE_MINIMUM_REQUIRED(VERSION 3.20)

#2. 确定工程名
PROJECT(TestTrader LANGUAGES CXX)


#7. 添加源码
SET(SRCS
	main.cpp
)

SET(LIBS
	WTSTools
	WTSUtils
	)
    
IF (MSVC)
ELSE()
	LIST(APPEND LIBS
		WT::Platform
		${FS_LIB})
    IF(WIN32)
		LIST(APPEND LIBS
			iconv)
	ENDIF()
ENDIF()


ADD_EXECUTABLE(TestTrader ${SRCS})
TARGET_LINK_LIBRARIES(TestTrader PRIVATE WT::fmt)
TARGET_LINK_LIBRARIES(TestTrader PRIVATE ${LIBS})

IF (MSVC)
ELSE()
	SET_TARGET_PROPERTIES(TestTrader PROPERTIES
        BUILD_RPATH_USE_ORIGIN ON)
ENDIF ()
