#1. 确定CMake的最低版本需求
CMAKE_MINIMUM_REQUIRED(VERSION 3.20)

#2. 确定工程名
PROJECT(ParserCTP LANGUAGES CXX)

SET(SRC  
	${PROJECT_SOURCE_DIR}/ParserCTP.cpp
	${PROJECT_SOURCE_DIR}/ParserCTP.h
)


ADD_LIBRARY(ParserCTP SHARED ${SRC})
TARGET_LINK_LIBRARIES(ParserCTP PRIVATE WT::fmt PUBLIC WT::CTP)

IF (MSVC)
ELSE()
	SET(LIBS
		WT::BoostThread
		${FS_LIB}
		WT::Platform
	)
	TARGET_LINK_LIBRARIES(ParserCTP PUBLIC ${LIBS})
ENDIF()

IF(WT_CTP_STATIC)
	TARGET_LINK_LIBRARIES(ParserCTP PRIVATE WT::CTP_MD)
ENDIF()

IF (MSVC)
ELSE()
	SET_TARGET_PROPERTIES(ParserCTP PROPERTIES
		CXX_VISIBILITY_PRESET hidden
		C_VISIBILITY_PRESET hidden
		VISIBILITY_INLINES_HIDDEN 1
        BUILD_RPATH_USE_ORIGIN ON)
ENDIF ()
