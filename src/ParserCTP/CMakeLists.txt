#1. 确定CMake的最低版本需求
CMAKE_MINIMUM_REQUIRED(VERSION 3.20)

#2. 确定工程名
PROJECT(ParserCTP LANGUAGES CXX)
SET(CMAKE_CXX_STANDARD 23)

SET(SRC  
	${PROJECT_SOURCE_DIR}/ParserCTP.cpp
	${PROJECT_SOURCE_DIR}/ParserCTP.h
)

SET(LIBRARY_OUTPUT_PATH ${CMAKE_BINARY_DIR}/build_${PLATFORM}/${CMAKE_BUILD_TYPE}/bin)

INCLUDE_DIRECTORIES(${INCS})
LINK_DIRECTORIES(${LNKS})
ADD_LIBRARY(ParserCTP SHARED ${SRC})
TARGET_INCLUDE_DIRECTORIES(ParserCTP PRIVATE ${WT_CTP_INCLUDE_DIRS})
IF(WT_CTP_STATIC)
	TARGET_COMPILE_DEFINITIONS(ParserCTP PRIVATE WT_CTP_STATIC)
ENDIF()

IF (MSVC)
ELSE (GNUCC)
	SET(LIBS
		boost_thread
		${FS_LIB}
		${WT_DL_LIB}
	)
	TARGET_LINK_LIBRARIES(ParserCTP ${LIBS})
ENDIF()

IF(WT_CTP_STATIC)
	TARGET_LINK_LIBRARIES(ParserCTP WT::CTP_MD)
ENDIF()

IF (MSVC)
ELSE (GNUCC)
	SET_TARGET_PROPERTIES(ParserCTP PROPERTIES
		CXX_VISIBILITY_PRESET hidden
		C_VISIBILITY_PRESET hidden
		VISIBILITY_INLINES_HIDDEN 1
        LINK_FLAGS_RELEASE -s)
ENDIF ()
