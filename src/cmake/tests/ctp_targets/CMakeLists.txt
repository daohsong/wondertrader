cmake_minimum_required(VERSION 3.20)
project(WonderTraderCtpTargets LANGUAGES CXX)

if(NOT WT_DEPENDENCIES_MODULE)
	message(FATAL_ERROR "WT_DEPENDENCIES_MODULE is required")
endif()

set(WT_CTP_HEADERS_FOUND ON)
set(WT_CTP_INCLUDE_DIRS "${CMAKE_CURRENT_LIST_DIR}/include")
set(WT_CTP_LIBRARIES_USABLE ON)
set(WT_CTP_MD_LIBRARY_RESOLVED "${CMAKE_COMMAND}")
set(WT_CTP_TRADER_LIBRARY_RESOLVED "${CMAKE_COMMAND}")

include("${WT_DEPENDENCIES_MODULE}")
WT_SETUP_CTP_TARGETS()

foreach(WT_EXPECTED_TARGET IN ITEMS WT::CTP WT::CTP_MD WT::CTP_TRADER)
	if(NOT TARGET ${WT_EXPECTED_TARGET})
		message(FATAL_ERROR "CTP provider did not create ${WT_EXPECTED_TARGET}")
	endif()
endforeach()

get_target_property(WT_CTP_INCLUDES WT_CTP INTERFACE_SYSTEM_INCLUDE_DIRECTORIES)
if(NOT "${CMAKE_CURRENT_LIST_DIR}/include" IN_LIST WT_CTP_INCLUDES)
	message(FATAL_ERROR "WT::CTP does not publish its headers as system includes")
endif()

foreach(WT_CTP_COMPONENT IN ITEMS MD TRADER)
	get_target_property(WT_CTP_COMPONENT_DEFINITIONS
		WT_CTP_${WT_CTP_COMPONENT}
		INTERFACE_COMPILE_DEFINITIONS
	)
	if(NOT "WT_CTP_STATIC" IN_LIST WT_CTP_COMPONENT_DEFINITIONS)
		message(FATAL_ERROR "WT::CTP_${WT_CTP_COMPONENT} does not publish WT_CTP_STATIC")
	endif()
	get_target_property(WT_CTP_COMPONENT_LINKS
		WT_CTP_${WT_CTP_COMPONENT}
		INTERFACE_LINK_LIBRARIES
	)
	foreach(WT_EXPECTED_LINK IN ITEMS WT::CTP WT_CTP_${WT_CTP_COMPONENT}_LIBRARY)
		if(NOT WT_EXPECTED_LINK IN_LIST WT_CTP_COMPONENT_LINKS)
			message(FATAL_ERROR "WT::CTP_${WT_CTP_COMPONENT} does not publish ${WT_EXPECTED_LINK}")
		endif()
	endforeach()
endforeach()
