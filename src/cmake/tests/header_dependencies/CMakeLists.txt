cmake_minimum_required(VERSION 3.20)
project(WonderTraderHeaderDependencies LANGUAGES CXX)

if(NOT WT_DEPENDENCIES_MODULE)
	message(FATAL_ERROR "WT_DEPENDENCIES_MODULE is required")
endif()

set(CMAKE_DISABLE_FIND_PACKAGE_fmt TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_RapidJSON TRUE)
set(WT_DEPS_ROOT "")
set(WT_FMT_INCLUDE_PATHS "${CMAKE_CURRENT_LIST_DIR}/include")
set(WT_RAPIDJSON_INCLUDE_PATHS "${CMAKE_CURRENT_LIST_DIR}/include")
set(WT_FMT_INCLUDE_DIR "${CMAKE_CURRENT_LIST_DIR}/include" CACHE PATH "" FORCE)
set(WT_RAPIDJSON_INCLUDE_DIR "${CMAKE_CURRENT_LIST_DIR}/include" CACHE PATH "" FORCE)

include("${WT_DEPENDENCIES_MODULE}")
WT_SETUP_HEADER_DEPENDENCIES()

if(NOT TARGET WT::fmt OR NOT TARGET WT::RapidJSON)
	message(FATAL_ERROR "Header dependency aliases were not created")
endif()

get_target_property(WT_FMT_DEFINITIONS WT_FMT INTERFACE_COMPILE_DEFINITIONS)
foreach(WT_REQUIRED_DEFINITION IN ITEMS WT_USE_STANDALONE_FMT FMT_HEADER_ONLY)
	if(NOT WT_REQUIRED_DEFINITION IN_LIST WT_FMT_DEFINITIONS)
		message(FATAL_ERROR "WT::fmt does not publish ${WT_REQUIRED_DEFINITION}")
	endif()
endforeach()

get_target_property(WT_FMT_INCLUDES WT_FMT INTERFACE_SYSTEM_INCLUDE_DIRECTORIES)
get_target_property(WT_RAPIDJSON_INCLUDES WT_RAPIDJSON INTERFACE_SYSTEM_INCLUDE_DIRECTORIES)
if(NOT "${CMAKE_CURRENT_LIST_DIR}/include" IN_LIST WT_FMT_INCLUDES)
	message(FATAL_ERROR "WT::fmt does not publish its include directory")
endif()
if(NOT "${CMAKE_CURRENT_LIST_DIR}/include" IN_LIST WT_RAPIDJSON_INCLUDES)
	message(FATAL_ERROR "WT::RapidJSON does not publish its include directory")
endif()
