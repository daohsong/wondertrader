cmake_minimum_required(VERSION 3.20)

PROJECT(TraderZMQ LANGUAGES CXX)

SET(SRC
	${PROJECT_SOURCE_DIR}/TraderZMQ.cpp
	${PROJECT_SOURCE_DIR}/TraderZMQ.h
)

SET(LIBRARY_OUTPUT_PATH ${CMAKE_BINARY_DIR}/build_${PLATFORM}/${CMAKE_BUILD_TYPE}/bin)

IF(WT_ZMQ_INCLUDE_DIR)
	SET(ZMQ_INCLUDE_DIR ${WT_ZMQ_INCLUDE_DIR})
ELSE()
	SET(TRADER_ZMQ_INCLUDE_PATHS ${INCS})
	IF(DEFINED WT_ZMQ_INCLUDE_PATHS)
		LIST(APPEND TRADER_ZMQ_INCLUDE_PATHS ${WT_ZMQ_INCLUDE_PATHS})
	ELSEIF(NOT APPLE)
		LIST(APPEND TRADER_ZMQ_INCLUDE_PATHS ${CMAKE_SOURCE_DIR}/../../wt_deps/include /usr/include)
	ENDIF()
	FIND_PATH(ZMQ_INCLUDE_DIR NAMES zmq.h PATHS
		${TRADER_ZMQ_INCLUDE_PATHS}
		NO_DEFAULT_PATH
	)
ENDIF()
IF(NOT ZMQ_INCLUDE_DIR)
	MESSAGE(FATAL_ERROR "zmq.h not found; please install zeromq dev headers or place them under wt_deps/include")
ENDIF()

IF(WT_RAPIDJSON_INCLUDE_DIR)
	SET(RAPIDJSON_INCLUDE_DIR ${WT_RAPIDJSON_INCLUDE_DIR})
ELSE()
	SET(TRADER_ZMQ_RAPIDJSON_INCLUDE_PATHS ${INCS})
	IF(DEFINED WT_RAPIDJSON_INCLUDE_PATHS)
		LIST(APPEND TRADER_ZMQ_RAPIDJSON_INCLUDE_PATHS ${WT_RAPIDJSON_INCLUDE_PATHS})
	ELSEIF(NOT APPLE)
		LIST(APPEND TRADER_ZMQ_RAPIDJSON_INCLUDE_PATHS
			${CMAKE_SOURCE_DIR}/../../wt_deps/include
			${PROJECT_SOURCE_DIR}/../../../wt_deps/include
			/usr/include
		)
	ENDIF()
	FIND_PATH(RAPIDJSON_INCLUDE_DIR NAMES rapidjson/document.h PATHS
		${TRADER_ZMQ_RAPIDJSON_INCLUDE_PATHS}
		NO_DEFAULT_PATH
	)
ENDIF()
IF(NOT RAPIDJSON_INCLUDE_DIR)
	MESSAGE(FATAL_ERROR "rapidjson not found; please install it or place headers under wt_deps/include/rapidjson")
ENDIF()

IF(WT_CPPZMQ_INCLUDE_DIR)
	SET(CPPZMQ_INCLUDE_DIR ${WT_CPPZMQ_INCLUDE_DIR})
ELSE()
	SET(TRADER_ZMQ_CPPZMQ_INCLUDE_PATHS ${INCS})
	IF(DEFINED WT_CPPZMQ_INCLUDE_PATHS)
		LIST(APPEND TRADER_ZMQ_CPPZMQ_INCLUDE_PATHS ${WT_CPPZMQ_INCLUDE_PATHS})
	ELSEIF(NOT APPLE)
		LIST(APPEND TRADER_ZMQ_CPPZMQ_INCLUDE_PATHS
			${CMAKE_SOURCE_DIR}/../../cppzmq
			${PROJECT_SOURCE_DIR}/../../../cppzmq
			/usr/include
		)
	ENDIF()
	FIND_PATH(CPPZMQ_INCLUDE_DIR NAMES zmq.hpp PATHS
		${TRADER_ZMQ_CPPZMQ_INCLUDE_PATHS}
		NO_DEFAULT_PATH
	)
ENDIF()
IF(NOT CPPZMQ_INCLUDE_DIR)
	MESSAGE(FATAL_ERROR "cppzmq (zmq.hpp) not found; please init submodule or configure WT_CPPZMQ_INCLUDE_DIR/WT_CPPZMQ_INCLUDE_PATHS")
ENDIF()

INCLUDE_DIRECTORIES(${INCS} ${RAPIDJSON_INCLUDE_DIR} ${ZMQ_INCLUDE_DIR} ${CPPZMQ_INCLUDE_DIR})
LINK_DIRECTORIES(${LNKS})
ADD_LIBRARY(TraderZMQ SHARED ${SRC})
IF (MSVC)
ELSE (GNUCC)
	TARGET_COMPILE_OPTIONS(TraderZMQ PRIVATE -isystem ${ZMQ_INCLUDE_DIR} -isystem ${CPPZMQ_INCLUDE_DIR})
ENDIF()

IF(WT_ZMQ_LIB)
	SET(ZMQ_LIB ${WT_ZMQ_LIB})
ELSE()
	SET(TRADER_ZMQ_LIBRARY_PATHS ${LNKS})
	IF(DEFINED WT_ZMQ_LIBRARY_PATHS)
		LIST(APPEND TRADER_ZMQ_LIBRARY_PATHS ${WT_ZMQ_LIBRARY_PATHS})
	ELSEIF(NOT APPLE)
		LIST(APPEND TRADER_ZMQ_LIBRARY_PATHS ${CMAKE_SOURCE_DIR}/../../wt_deps/lib /usr/lib /usr/lib/x86_64-linux-gnu)
	ENDIF()
	FIND_LIBRARY(ZMQ_LIB NAMES zmq libzmq PATHS
		${TRADER_ZMQ_LIBRARY_PATHS}
		NO_DEFAULT_PATH
	)
ENDIF()
IF(NOT ZMQ_LIB)
	MESSAGE(FATAL_ERROR "libzmq not found; please install zeromq or place it under wt_deps/lib")
ENDIF()

IF (MSVC)
ELSE (GNUCC)
	SET(LIBS
		${ZMQ_LIB}
		${WT_THREAD_LIB}
		${WT_DL_LIB}
	)
	TARGET_LINK_LIBRARIES(TraderZMQ ${LIBS})
ENDIF()

IF (MSVC)
ELSE (GNUCC)
	SET_TARGET_PROPERTIES(TraderZMQ PROPERTIES
		CXX_VISIBILITY_PRESET hidden
		C_VISIBILITY_PRESET hidden
		VISIBILITY_INLINES_HIDDEN 1)
ENDIF()
