
#1. 确定CMake的最低版本需求
CMAKE_MINIMUM_REQUIRED(VERSION 3.20)

#2. 确定工程名
PROJECT(TraderCTP LANGUAGES CXX)

SET(SRC  
	${PROJECT_SOURCE_DIR}/TraderCTP.cpp
	${PROJECT_SOURCE_DIR}/TraderCTP.h
)


ADD_LIBRARY(TraderCTP SHARED ${SRC})
TARGET_LINK_LIBRARIES(TraderCTP PRIVATE WT::fmt PUBLIC WT::CTP)

IF(MSVC)
ELSE()
	SET(LIBS
		WT::Platform
		${FS_LIB}
        WT::BoostFilesystem
		WT::Platform
	)
	TARGET_LINK_LIBRARIES(TraderCTP PUBLIC ${LIBS})
ENDIF()

IF(WT_CTP_STATIC)
	TARGET_LINK_LIBRARIES(TraderCTP PRIVATE WT::CTP_TRADER)
ENDIF()

IF (MSVC)
ELSE()
	SET_TARGET_PROPERTIES(TraderCTP PROPERTIES
		CXX_VISIBILITY_PRESET hidden
		C_VISIBILITY_PRESET hidden
		VISIBILITY_INLINES_HIDDEN 1
        BUILD_RPATH_USE_ORIGIN ON)
ENDIF ()
