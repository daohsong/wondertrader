#1. 确定CMake的最低版本需求
CMAKE_MINIMUM_REQUIRED(VERSION 3.20)

#2. 确定工程名
PROJECT(TraderCTPOpt LANGUAGES CXX)

SET(SRC  
	${PROJECT_SOURCE_DIR}/TraderCTPOpt.cpp
	${PROJECT_SOURCE_DIR}/TraderCTPOpt.h
)


ADD_LIBRARY(TraderCTPOpt SHARED ${SRC})
TARGET_LINK_LIBRARIES(TraderCTPOpt PRIVATE WT::fmt PUBLIC WT::Vendor::CTPOpt)

IF(MSVC)
ELSE()
	SET(LIBS
		WT::Platform
		${FS_LIB}
		WT::Platform
		)
	TARGET_LINK_LIBRARIES(TraderCTPOpt PUBLIC ${LIBS})
ENDIF()

IF (MSVC)
ELSE()
	SET_TARGET_PROPERTIES(TraderCTPOpt PROPERTIES
		CXX_VISIBILITY_PRESET hidden
		C_VISIBILITY_PRESET hidden
		VISIBILITY_INLINES_HIDDEN 1
        BUILD_RPATH_USE_ORIGIN ON)
ENDIF ()
