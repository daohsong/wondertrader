
#1. 确定CMake的最低版本需求
CMAKE_MINIMUM_REQUIRED(VERSION 3.20)

#2. 确定工程名
PROJECT(TestPorter LANGUAGES CXX)

SET(EXECUTABLE_OUTPUT_PATH ${CMAKE_BINARY_DIR}/build_${PLATFORM}/${CMAKE_BUILD_TYPE}/bin/WtPorter)

#7. 添加源码
file(GLOB SRCS *.cpp)

SET(LIBS
	WtPorter
)


ADD_EXECUTABLE(TestPorter ${SRCS})
TARGET_LINK_LIBRARIES(TestPorter PRIVATE WT::fmt WT::Boost)
TARGET_LINK_LIBRARIES(TestPorter PRIVATE ${LIBS})

IF (MSVC)
ELSE (GNUCC)
	SET_TARGET_PROPERTIES(TestPorter PROPERTIES
        LINK_FLAGS_RELEASE -s)
ENDIF ()
